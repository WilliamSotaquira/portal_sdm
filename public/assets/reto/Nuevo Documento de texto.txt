    <script>
        const menuToggle = document.querySelector('.reto-menu-toggle');
        const menu = document.querySelector('.reto-menu');
        const inscription = document.querySelector('.nav-inscription');

        menuToggle.addEventListener('click', () => {
            const expanded = menuToggle.getAttribute('aria-expanded') === 'true' || false;
            menuToggle.setAttribute('aria-expanded', !expanded);
            menu.style.display = !expanded ? 'flex' : 'none';
            inscription.style.display = !expanded ? 'flex' : 'none';
        });

        // ubicar correctamente el menú desplegable en dispositivos móviles
        window.addEventListener('resize', () => {
            if (window.innerWidth > 768) {
                menu.style.display = 'flex';
                inscription.style.display = 'flex';
                menuToggle.setAttribute('aria-expanded', 'false');
            } else {
                menu.style.display = 'none';
                inscription.style.display = 'none';
                menuToggle.setAttribute('aria-expanded', 'false');
            }
        });

        // reubicar correctamente el boton de inscripción dentro del menu (nav-link-small) abrir el menú en dispositivos móviles
        menuToggle.addEventListener('click', () => {
            if (window.innerWidth <= 768) {
                const navLinks = document.querySelectorAll('.nav-link-small');
                const lastNavLink = navLinks[navLinks.length - 1];
                lastNavLink.insertAdjacentElement('afterend', inscription);
            }
        });
    </script>
